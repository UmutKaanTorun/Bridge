> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Amaç:** Faz 0'daki Wine betiklerini (`oyun26.sh`, `steam26.sh`, `steam-kur.sh`), CrossOver modelinde çalışan bir Mac uygulamasına (Köprü) dönüştürmek: tarifle kurulum (Steam), listede olmayan program kurulumu, bottle ve program yönetimi, var olan bottle'ı içe aktarma.

- macOS 14.0+, yalnızca Apple Silicon (`ARCHS = arm64`). Motor x86_64'tür, Rosetta ile çalışır.
- Swift 6 dil modu: paket `swift-tools-version: 6.0`, uygulama `SWIFT_VERSION = 6.0`.
- Görünen ad **Köprü**; dosya, klasör, hedef ve kod adı **Kopru**; paket `Kopru.app`; bundle ID `dev.kopru.Kopru` (geçici).

```swift
import Foundation
@testable import KopruCore

/// Test başına geçici klasör; nesne bırakılınca silinir.
final class TempDir: @unchecked Sendable {
    let url: URL

    init() throws {
        url = FileManager.default.temporaryDirectory
            .appendingPathComponent("KopruTests-\(UUID().uuidString)", isDirectory: true)
        try FileManager.default.createDirectory(at: url, withIntermediateDirectories: true)
    }

    deinit { try? FileManager.default.removeItem(at: url) }

    /// Göreli yolda (ara klasörleriyle) metin dosyası oluşturur.
    @discardableResult
    func file(_ relative: String, _ contents: String = "") throws -> URL {
        try file(relative, data: Data(contents.utf8))
    }

    @discardableResult
    func file(_ relative: String, data: Data) throws -> URL {
        let target = url.appendingPathComponent(relative)
        try FileManager.default.createDirectory(at: target.deletingLastPathComponent(), withIntermediateDirectories: true)
        try data.write(to: target)
        return target
    }

    @discardableResult
    func dir(_ relative: String) throws -> URL {
        let target = url.appendingPathComponent(relative, isDirectory: true)
        try FileManager.default.createDirectory(at: target, withIntermediateDirectories: true)
        return target
    }

    /// Bu klasörün altındaki Köprü yolları.
    var paths: KopruPaths {
        KopruPaths(applicationSupport: url.appendingPathComponent("AppSupport", isDirectory: true),
                   logs: url.appendingPathComponent("Logs", isDirectory: true))
    }
}
```

```swift
import Foundation

/// Köprü'nün diskteki klasörleri. Testler geçici bir kök verir.
public struct KopruPaths: Sendable, Equatable {
    /// ~/Library/Application Support/Kopru
    public let applicationSupport: URL
    /// ~/Library/Logs/Kopru
    public let logs: URL

    public init(applicationSupport: URL, logs: URL) {
        self.applicationSupport = applicationSupport
        self.logs = logs
    }

    public static var standard: KopruPaths {
        let library = FileManager.default.urls(for: .libraryDirectory, in: .userDomainMask)[0]
        return KopruPaths(
            applicationSupport: library.appendingPathComponent("Application Support/Kopru", isDirectory: true),
            logs: library.appendingPathComponent("Logs/Kopru", isDirectory: true)
        )
    }

    /// Her alt klasör bir Wine bottle'ı (WINEPREFIX).
    public var bottles: URL { applicationSupport.appendingPathComponent("Bottles", isDirectory: true) }

    /// İndirilen kurulum dosyaları.
    public var downloads: URL { applicationSupport.appendingPathComponent("Downloads", isDirectory: true) }

    public func logFile(forBottleFolder folder: String) -> URL {
        logs.appendingPathComponent("\(folder).log")
    }
}
```

```bash
# Release derlemesinde Wine motorunu Kopru.app/Contents/Resources/Motor'a kopyalar.

  exit 1

echo "Motor kopyalandı: $DST"
```

```swift
import Foundation
import Testing
@testable import KopruCore

struct BottleSettingsTests {
    @Test func autoDesktopIsScreenMinus40() {
        let size = VirtualDesktop.auto.size(visibleScreen: PixelSize(width: 1512, height: 944))
        #expect(size == PixelSize(width: 1472, height: 904))
    }

    @Test func offDesktopHasNoSize() {
        #expect(VirtualDesktop.off.size(visibleScreen: PixelSize(width: 1512, height: 944)) == nil)
    }

    @Test func fixedDesktopKeepsItsSize() {
        let size = VirtualDesktop.fixed(width: 1280, height: 800).size(visibleScreen: PixelSize(width: 1512, height: 944))
        #expect(size == PixelSize(width: 1280, height: 800))
    }

    @Test func decodesDesktopModes() throws {
        let decoder = JSONDecoder()
        #expect(try decoder.decode(VirtualDesktop.self, from: Data(#"{"mode":"auto"}"#.utf8)) == .auto)
        #expect(try decoder.decode(VirtualDesktop.self, from: Data(#"{"mode":"off"}"#.utf8)) == .off)
        #expect(try decoder.decode(VirtualDesktop.self, from: Data(#"{"mode":"fixed","width":1280,"height":800}"#.utf8))
                == .fixed(width: 1280, height: 800))
        #expect(throws: DecodingError.self) {
            try decoder.decode(VirtualDesktop.self, from: Data(#"{"mode":"yan"}"#.utf8))
        }
    }

    @Test func settingsRoundTrip() throws {
        let settings = BottleSettings(msync: false, virtualDesktop: .fixed(width: 1024, height: 768), fpsHud: true)
        let data = try JSONEncoder().encode(settings)
        #expect(try JSONDecoder().decode(BottleSettings.self, from: data) == settings)
    }

    @Test func defaultSettings() {
        #expect(BottleSettings.default == BottleSettings(msync: true, virtualDesktop: .off, fpsHud: false))
    }
}
```

```swift
import Foundation

/// Görünen bottle adından diskte kullanılacak ASCII klasör adı üretir.
/// Bazı Windows kurulum programları Türkçe karakterli yollarda sorun çıkarabiliyor.
public enum FolderName {
    private static let separators = CharacterSet.whitespaces.union(CharacterSet(charactersIn: "/:\\"))
    private static let extraAllowed = CharacterSet(charactersIn: "-_+.")

    public static func make(from displayName: String, existing: Set<String>) -> String {
        // Noktasız ı ve noktalı İ aksan değil; elle çevrilir. Diğer Türkçe harfler aksan atılarak çevrilir.
        let mapped = displayName.replacingOccurrences(of: "ı", with: "i").replacingOccurrences(of: "İ", with: "I")
        let stripped = mapped.applyingTransform(.stripDiacritics, reverse: false) ?? mapped

        var result = ""
        for scalar in stripped.unicodeScalars {
            if separators.contains(scalar) {
                result += "-"
            } else if scalar.isASCII && (CharacterSet.alphanumerics.contains(scalar) || extraAllowed.contains(scalar)) {
                result.unicodeScalars.append(scalar)
            }
        }
        while result.contains("--") {
            result = result.replacingOccurrences(of: "--", with: "-")
        }
        result = result.trimmingCharacters(in: CharacterSet(charactersIn: "-."))
        if result.isEmpty { result = "Bottle" }

        // APFS büyük/küçük harf duyarsız: çakışma kontrolü de öyle.
        let taken = Set(existing.map { $0.lowercased() })
        guard taken.contains(result.lowercased()) else { return result }
        var number = 2
        while taken.contains("\(result)-\(number)".lowercased()) { number += 1 }
        return "\(result)-\(number)"
    }
}
```

```swift
import Foundation

/// Uygulamanın içindeki tarifler.
public struct RecipeCatalog: Sendable {
    public let recipes: [Recipe]
    /// Tariflere ait ek dosyalar: <filesDirectory>/<tarif id>/...
    public let filesDirectory: URL

    public init(recipes: [Recipe], filesDirectory: URL) {
        self.recipes = recipes.sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
        self.filesDirectory = filesDirectory
    }

    /// Klasördeki bütün .json tariflerini okur.
    public static func load(recipesDirectory: URL, filesDirectory: URL) throws -> RecipeCatalog {
        let files = try FileManager.default
            .contentsOfDirectory(at: recipesDirectory, includingPropertiesForKeys: nil)
            .filter { $0.pathExtension.lowercased() == "json" }
        let decoder = JSONDecoder()
        let recipes = try files.map { try decoder.decode(Recipe.self, from: Data(contentsOf: $0)) }
        return RecipeCatalog(recipes: recipes, filesDirectory: filesDirectory)
    }

    /// Ad ve özette büyük/küçük harf ve aksan duyarsız arama; boş sorgu hepsini döndürür.
    public func search(_ query: String) -> [Recipe] {
        let q = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !q.isEmpty else { return recipes }
        let options: String.CompareOptions = [.caseInsensitive, .diacriticInsensitive]
        return recipes.filter {
            $0.name.range(of: q, options: options) != nil || $0.summary.range(of: q, options: options) != nil
        }
    }

    public func recipe(id: String?) -> Recipe? {
        guard let id else { return nil }
        return recipes.first { $0.id == id }
    }

    public func filesDirectory(for recipe: Recipe) -> URL {
        filesDirectory.appendingPathComponent(recipe.id, isDirectory: true)
    }

    /// Bottle'da `detect` dosyalarının hepsi bulunan ilk tarif.
    public func detect(in bottleURL: URL) -> Recipe? {
        recipes.first { recipe in
            !recipe.detect.isEmpty && recipe.detect.allSatisfy {
                FileManager.default.fileExists(atPath: bottleURL.appendingPathComponent($0).path)
            }
        }
    }
}
```

```swift
import Foundation

public enum EngineError: Error, Equatable, LocalizedError {
    case notFound(searched: [String])
    case rosettaMissing

    public var errorDescription: String? {
        switch self {
        case let .notFound(searched):
            "Wine motoru bulunamadı. Aranan yerler:\n" + searched.joined(separator: "\n")
        case .rosettaMissing:
            "Köprü'nün çalışması için Rosetta gerekiyor. Terminal'de şu komutu çalıştır:\nsoftwareupdate --install-rosetta --agree-to-license"
        }
    }
}

/// CrossOver 26.3.0 kaynağından derlenmiş Wine 11.0 + DXMT motoru. `oyun26.sh`'in karşılığı.
public struct Engine: Sendable, Equatable {
    public let root: URL

    public init(root: URL) { self.root = root }

    public var wine: URL { root.appendingPathComponent("bin/wine") }
    public var wineserver: URL { root.appendingPathComponent("bin/wineserver") }
    public var libraries: URL { root.appendingPathComponent("lib", isDirectory: true) }

    /// Geliştirme sırasında kullanılan motor (Faz 0'da derlendi).
    public static let developmentEngine = FileManager.default.homeDirectoryForCurrentUser
        .appendingPathComponent("WineEngines/WineCX26.3.0-DXMT0.80", isDirectory: true)

    public static let rosettaRuntimePath = "/Library/Apple/usr/libexec/oah/libRosettaRuntime"

    public static func isValid(_ root: URL) -> Bool {
        FileManager.default.isExecutableFile(atPath: root.appendingPathComponent("bin/wine").path)
    }

    /// Arama sırası: uygulamanın içi (Resources/Motor) → KOPRU_MOTOR → geliştirme motoru.
    public static func locate(bundleResources: URL?, environment: [String: String], developmentFallback: URL?) throws -> Engine {
        var candidates: [URL] = []
        if let bundleResources {
            candidates.append(bundleResources.appendingPathComponent("Motor", isDirectory: true))
        }
        if let custom = environment["KOPRU_MOTOR"], !custom.isEmpty {
            candidates.append(URL(fileURLWithPath: custom, isDirectory: true))
        }
        if let developmentFallback {
            candidates.append(developmentFallback)
        }
        guard let found = candidates.first(where: isValid) else {
            throw EngineError.notFound(searched: candidates.map(\.path))
        }
        return Engine(root: found)
    }

    /// Motor x86_64; Apple Silicon'da Rosetta olmadan açılmaz.
    public static func checkRosetta(fileExists: (String) -> Bool = { FileManager.default.fileExists(atPath: $0) }) throws {
        guard fileExists(rosettaRuntimePath) else { throw EngineError.rosettaMissing }
    }

    /// Wine'a verilecek ortam değişkenleri: `base`'in üzerine Köprü'nün değerleri yazılır.
    public func environment(prefix: URL, settings: BottleSettings,
                            base: [String: String] = ProcessInfo.processInfo.environment) -> [String: String] {
        var env = base
        env["WINEPREFIX"] = prefix.path
        env["DYLD_FALLBACK_LIBRARY_PATH"] = libraries.path
        env["WINEMSYNC"] = settings.msync ? "1" : "0"
        env["MVK_CONFIG_LOG_LEVEL"] = "0"
        env["WINEDEBUG"] = base["WINEDEBUG"] ?? "-all"
        env["MTL_HUD_ENABLED"] = settings.fpsHud ? "1" : nil
        return env
    }
}
```

```swift
import Foundation

/// Wine'a ne çalıştırılacağı.
public enum LaunchTarget: Hashable, Sendable {
    /// Bottle içindeki Windows yolu: `wine C:\…\x.exe args`
    case windowsPath(String, arguments: [String])
    /// Başlat menüsü kısayolu (.lnk/.url): `wine start /wait /unix <dosya>`;
    /// /wait sayesinde süreç program kapanana kadar yaşar.
    case shortcut(URL)
    /// Mac'teki bir dosya: .exe doğrudan, .msi msiexec ile.
    case unixFile(URL, arguments: [String])
    /// Wine'ın kendi programları: wineboot, winecfg, reg, cmd…
    case builtin(String, arguments: [String])
}

/// Wine sanal masaüstü penceresi.
public struct DesktopWindow: Hashable, Sendable {
    public var name: String
    public var size: PixelSize

    public init(name: String, size: PixelSize) {
        self.name = name
        self.size = size
    }
}

public enum WineArguments {
    /// `wine` ikilisine verilecek argümanlar. Sanal masaüstü varsa komut explorer'a sarılır.
    public static func arguments(for target: LaunchTarget, desktop: DesktopWindow?) -> [String] {
        let inner: [String]
        switch target {
        case let .windowsPath(path, arguments):
            inner = [path] + arguments
        case let .shortcut(file):
            inner = ["start", "/wait", "/unix", file.path]
        case let .unixFile(file, arguments):
            inner = file.pathExtension.lowercased() == "msi"
                ? ["msiexec", "/i", file.path] + arguments
                : [file.path] + arguments
        case let .builtin(name, arguments):
            inner = [name] + arguments
        }
        guard let desktop else { return inner }
        return ["explorer", "/desktop=\(desktop.name),\(desktop.size.width)x\(desktop.size.height)"] + inner
    }
}
```

```swift
import Foundation

/// Wine'ın Windows yolları ile Mac yolları arasında çeviri. C: = <bottle>/drive_c, Z: = Mac'in kökü.
public enum WindowsPath {
    public static func toUnix(_ windowsPath: String, bottleURL: URL) -> URL? {
        let normalized = windowsPath.replacingOccurrences(of: "\\", with: "/")
        let characters = Array(normalized)
        guard characters.count >= 2, characters[1] == ":" else { return nil }
        let rest = String(characters.dropFirst(2)).trimmingCharacters(in: CharacterSet(charactersIn: "/"))
        switch characters[0].lowercased() {
        case "c":
            let driveC = bottleURL.appendingPathComponent("drive_c", isDirectory: true)
            return rest.isEmpty ? driveC : driveC.appendingPathComponent(rest)
        case "z":
            return URL(fileURLWithPath: "/" + rest)
        default:
            return nil
        }
    }

    public static func fromUnix(_ url: URL, bottleURL: URL) -> String {
        let driveC = bottleURL.appendingPathComponent("drive_c", isDirectory: true).standardizedFileURL.path
        let path = url.standardizedFileURL.path
        if path == driveC { return "C:\\" }
        if path.hasPrefix(driveC + "/") {
            return "C:\\" + String(path.dropFirst(driveC.count + 1)).replacingOccurrences(of: "/", with: "\\")
        }
        return "Z:" + path.replacingOccurrences(of: "/", with: "\\")
    }
}
```

```swift
import Foundation

public enum BottleStoreError: Error, Equatable, LocalizedError {
    case emptyName
    case nameTaken(String)
    case notAWineBottle(String)

    public var errorDescription: String? {
        switch self {
        case .emptyName: "Bottle adı boş olamaz."
        case let .nameTaken(name): "“\(name)” adında bir bottle zaten var."
        case let .notAWineBottle(path): "Bu klasör bir Wine bottle'ı değil (içinde system.reg ve drive_c olmalı): \(path)"
        }
    }
}

/// ~/Library/Application Support/Kopru/Bottles altındaki bottle'lar.
public struct BottleStore: Sendable {
    public let paths: KopruPaths

    public init(paths: KopruPaths) { self.paths = paths }

    /// kopru.json'u olan klasörler, görünen ada göre sıralı.
    public func list() throws -> [Bottle] {
        let fm = FileManager.default
        guard fm.fileExists(atPath: paths.bottles.path) else { return [] }
        let folders = try fm.contentsOfDirectory(at: paths.bottles, includingPropertiesForKeys: nil, options: [.skipsHiddenFiles])
        return folders
            .compactMap { url in (try? BottleInfo.read(from: url)).map { Bottle(url: url, info: $0) } }
            .sorted { $0.info.name.localizedStandardCompare($1.info.name) == .orderedAscending }
    }

    /// Kırpılmış adı döndürür; boşsa ya da başka bir bottle'da varsa hata.
    public func validateNewName(_ name: String, excluding: Bottle? = nil) throws -> String {
        let trimmed = name.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { throw BottleStoreError.emptyName }
        let taken = try list().contains {
            $0.id != excluding?.id && $0.info.name.compare(trimmed, options: .caseInsensitive) == .orderedSame
        }
        if taken { throw BottleStoreError.nameTaken(trimmed) }
        return trimmed
    }

    /// Klasörü ve kopru.json'u oluşturur. Windows ortamını (wineboot) kurmaz.
    public func makeBottle(name: String, settings: BottleSettings, recipeId: String?, now: Date = Date()) throws -> Bottle {
        let name = try validateNewName(name)
        let folder = FolderName.make(from: name, existing: try existingFolderNames())
        let url = paths.bottles.appendingPathComponent(folder, isDirectory: true)
        try FileManager.default.createDirectory(at: url, withIntermediateDirectories: true)
        let bottle = Bottle(url: url, info: BottleInfo(name: name, createdAt: now, recipeId: recipeId, settings: settings))
        try save(bottle)
        return bottle
    }

    public func save(_ bottle: Bottle) throws {
        try bottle.info.write(to: bottle.url)
    }

    /// Yalnızca görünen ad değişir; klasör adı aynı kalır.
    public func rename(_ bottle: Bottle, to newName: String) throws -> Bottle {
        var renamed = bottle
        renamed.info.name = try validateNewName(newName, excluding: bottle)
        try save(renamed)
        return renamed
    }

    /// Varsayılan olarak Çöp Kutusu'na taşır.
    public func delete(_ bottle: Bottle,
                       remover: (URL) throws -> Void = { try FileManager.default.trashItem(at: $0, resultingItemURL: nil) }) throws {
        try remover(bottle.url)
    }

    public static func isWineBottle(_ url: URL) -> Bool {
        let fm = FileManager.default
        return fm.fileExists(atPath: url.appendingPathComponent("system.reg").path)
            && fm.fileExists(atPath: url.appendingPathComponent("drive_c").path)
    }

    /// Var olan bir Wine bottle'ını Bottles klasörüne taşır ve tanıtır.
    /// Çağıran, taşımadan önce o bottle'ın wineserver'ını kapatmalıdır.
    public func importBottle(from source: URL, name: String, catalog: RecipeCatalog?, now: Date = Date()) throws -> Bottle {
        guard Self.isWineBottle(source) else { throw BottleStoreError.notAWineBottle(source.path) }
        let name = try validateNewName(name)
        let fm = FileManager.default
        try fm.createDirectory(at: paths.bottles, withIntermediateDirectories: true)
        let folder = FolderName.make(from: name, existing: try existingFolderNames())
        let destination = paths.bottles.appendingPathComponent(folder, isDirectory: true)
        try fm.moveItem(at: source, to: destination)

        let recipe = catalog?.detect(in: destination)
        let settings = recipe?.bottle.settings ?? BottleSettings(msync: true, virtualDesktop: .off, fpsHud: false)
        let bottle = Bottle(url: destination, info: BottleInfo(name: name, createdAt: now, recipeId: recipe?.id, settings: settings))
        try save(bottle)
        return bottle
    }

    private func existingFolderNames() throws -> Set<String> {
        let fm = FileManager.default
        guard fm.fileExists(atPath: paths.bottles.path) else { return [] }
        return Set(try fm.contentsOfDirectory(atPath: paths.bottles.path))
    }
}
```

```swift
import Foundation

/// Başlat menüsünde bulunan program kısayolu.
public struct ShortcutProgram: Hashable, Sendable, Identifiable {
    public enum Kind: Hashable, Sendable {
        /// .lnk kısayolu
        case link
        /// URL=steam://… içeren .url kısayolu (Steam oyunu)
        case steamURL(String)
    }

    public let name: String
    /// Programs klasörüne göre göreli yol, ör. "Steam/Steam.lnk".
    public let relativePath: String
    public let url: URL
    public let kind: Kind

    public var id: String { relativePath }
}

/// Bottle'ın Başlat menüsündeki kısayolları tarar.
public enum ProgramScanner {
    static let programsSubpath = "Microsoft/Windows/Start Menu/Programs"
    /// Adında bunlar geçen kısayollar program değil (kaldırıcı, belge).
    static let excludedWords = ["uninstall", "kaldır", "readme", "benioku", "yardım", "help", "kılavuz", "manual"]

    /// Ortak Başlat menüsü + her Wine kullanıcısının Başlat menüsü (CrossOver motorunda kullanıcı "crossover").
    public static func roots(bottleURL: URL) -> [URL] {
        let driveC = bottleURL.appendingPathComponent("drive_c", isDirectory: true)
        var roots = [driveC.appendingPathComponent("ProgramData").appendingPathComponent(programsSubpath)]
        let users = driveC.appendingPathComponent("users", isDirectory: true)
        let names = (try? FileManager.default.contentsOfDirectory(atPath: users.path)) ?? []
        for name in names.sorted() {
            roots.append(users.appendingPathComponent(name).appendingPathComponent("AppData/Roaming").appendingPathComponent(programsSubpath))
        }
        return roots
    }

    /// Kısayollar ada göre sıralı; aynı göreli yol bir kez alınır.
    public static func scan(bottleURL: URL, hidden: Set<String>) -> [ShortcutProgram] {
        var seen = Set<String>()
        var result: [ShortcutProgram] = []
        for root in roots(bottleURL: bottleURL) {
            guard let enumerator = FileManager.default.enumerator(atPath: root.path) else { continue }
            while let relative = enumerator.nextObject() as? String {
                let ext = (relative as NSString).pathExtension.lowercased()
                guard ext == "lnk" || ext == "url", !seen.contains(relative), !hidden.contains(relative) else { continue }
                let name = ((relative as NSString).lastPathComponent as NSString).deletingPathExtension
                if excludedWords.contains(where: { name.localizedCaseInsensitiveContains($0) }) { continue }
                let url = root.appendingPathComponent(relative)
                let kind: ShortcutProgram.Kind
                if ext == "url" {
                    guard let target = steamURL(in: url) else { continue }
                    kind = .steamURL(target)
                } else {
                    kind = .link
                }
                seen.insert(relative)
                result.append(ShortcutProgram(name: name, relativePath: relative, url: url, kind: kind))
            }
        }
        return result.sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }

    /// .url dosyasındaki "URL=steam://…" değeri; Steam dışı bağlantılar için nil.
    static func steamURL(in file: URL) -> String? {
        guard let data = try? Data(contentsOf: file) else { return nil }
        for line in String(decoding: data, as: UTF8.self).split(whereSeparator: \.isNewline) {
            let trimmed = line.trimmingCharacters(in: .whitespaces)
            guard trimmed.lowercased().hasPrefix("url=") else { continue }
            let value = String(trimmed.dropFirst(4))
            return value.lowercased().hasPrefix("steam://") ? value : nil
        }
        return nil
    }
}
```

```swift
/// Program ızgarasındaki bir kutucuk.
public struct ProgramItem: Identifiable, Hashable, Sendable {
    public enum Source: Hashable, Sendable {
        case recipe(RecipeLaunch)
        case pinned(PinnedProgram)
        case shortcut(ShortcutProgram)
    }

    public let name: String
    public let source: Source
    public let target: LaunchTarget

    public var id: String {
        switch source {
        case let .recipe(launch): "recipe:\(launch.windowsPath)"
        case let .pinned(program): "pinned:\(program.windowsPath)"
        case let .shortcut(shortcut): "shortcut:\(shortcut.relativePath)"
        }
    }
}

public enum ProgramList {
    /// Sıra: tarif başlatıcısı, sabitlenenler, kısayollar.
    /// Steam oyunu kısayolları, tarif varsa tarifin başlatıcısıyla açılır (aracı ve seçenekler atlanmasın).
    public static func items(recipeLaunch: RecipeLaunch?, pinned: [PinnedProgram], shortcuts: [ShortcutProgram]) -> [ProgramItem] {
        var items: [ProgramItem] = []
        if let launch = recipeLaunch {
            items.append(ProgramItem(name: launch.name, source: .recipe(launch),
                                     target: .windowsPath(launch.windowsPath, arguments: launch.arguments)))
        }
        for program in pinned {
            items.append(ProgramItem(name: program.name, source: .pinned(program),
                                     target: .windowsPath(program.windowsPath, arguments: program.arguments)))
        }
        for shortcut in shortcuts {
            let target: LaunchTarget
            if case let .steamURL(url) = shortcut.kind, let launch = recipeLaunch {
                target = .windowsPath(launch.windowsPath, arguments: launch.arguments + [url])
            } else {
                target = .shortcut(shortcut.url)
            }
            items.append(ProgramItem(name: shortcut.name, source: .shortcut(shortcut), target: target))
        }
        return items
    }
```

```swift
public struct RunResult: Equatable, Sendable {
    public let exitCode: Int32

/// Wine süreçlerini başlatır, çıktıyı bottle'ın günlüğüne yazar, bottle'ı durdurur.
public final class WineRunner: @unchecked Sendable {
    public let engine: Engine
    public let paths: KopruPaths
    private let baseEnvironment: [String: String]
    private let lock = NSLock()
    private var counts: [String: Int] = [:]

    public init(engine: Engine, paths: KopruPaths, baseEnvironment: [String: String] = ProcessInfo.processInfo.environment) {
        self.engine = engine
        self.paths = paths
        self.baseEnvironment = baseEnvironment
    }

    /// Köprü'nün bu bottle'da başlattığı ve henüz bitmemiş süreç var mı.
    public func isRunning(_ bottle: Bottle) -> Bool {
        lock.withLock { (counts[bottle.folderName] ?? 0) > 0 }
    }

    /// Programı başlatır ve bitmesini bekler.
    public func run(_ target: LaunchTarget, in bottle: Bottle, desktopSize: PixelSize? = nil,
                    extraEnvironment: [String: String] = [:]) async throws -> RunResult {
        let desktop = desktopSize.map { DesktopWindow(name: "Kopru-\(bottle.folderName)", size: $0) }
        let arguments = WineArguments.arguments(for: target, desktop: desktop)

        let process = Process()
        process.executableURL = engine.wine
        process.arguments = arguments
        process.environment = engine.environment(prefix: bottle.url, settings: bottle.info.settings, base: baseEnvironment)
            .merging(extraEnvironment) { _, new in new }
        if case let .unixFile(file, _) = target {
            process.currentDirectoryURL = file.deletingLastPathComponent()
        } else {
            process.currentDirectoryURL = bottle.url
        }

        let log = try openLog(for: bottle)
        log.write(Data("\n=== \(Date().formatted(.iso8601)) wine \(arguments.joined(separator: " "))\n".utf8))
        process.standardOutput = log
        process.standardError = log

        let key = bottle.folderName
        let start = Date()
        adjust(key, by: 1)
        return try await withCheckedThrowingContinuation { continuation in
            process.terminationHandler = { [self] finished in
                adjust(key, by: -1)
                try? log.close()
                continuation.resume(returning: RunResult(exitCode: finished.terminationStatus,
                                                         duration: Date().timeIntervalSince(start)))
            }
            do {
                try process.run()
            } catch {
                adjust(key, by: -1)
                try? log.close()
                continuation.resume(throwing: error)
            }
        }
    }

    /// `wineserver -k`: bottle'daki bütün Windows süreçlerini kapatır.
    public func killWineserver(prefix: URL, settings: BottleSettings) async {
        let process = Process()
        process.executableURL = engine.wineserver
        process.arguments = ["-k"]
        process.environment = engine.environment(prefix: prefix, settings: settings, base: baseEnvironment)
        process.standardOutput = FileHandle.nullDevice
        process.standardError = FileHandle.nullDevice
        await withCheckedContinuation { (continuation: CheckedContinuation<Void, Never>) in
            process.terminationHandler = { _ in continuation.resume() }
            do { try process.run() } catch { continuation.resume() }
        }
    }

    /// Önce programı kibarca kapatmayı dener (`shutdown`), `grace` kadar bekler, sonra wineserver'ı öldürür.
    public func stop(_ bottle: Bottle, shutdown: LaunchTarget? = nil, grace: Duration = .seconds(10)) async {
        if let shutdown {
            Task { _ = try? await self.run(shutdown, in: bottle) }
            let clock = ContinuousClock()
            let deadline = clock.now + grace
            try? await Task.sleep(for: .milliseconds(300))
            while clock.now < deadline && isRunning(bottle) {
                try? await Task.sleep(for: .milliseconds(300))
            }
        }
        await killWineserver(prefix: bottle.url, settings: bottle.info.settings)
    }

    func openLog(for bottle: Bottle) throws -> FileHandle {
        let fm = FileManager.default
        try fm.createDirectory(at: paths.logs, withIntermediateDirectories: true)
        let url = paths.logFile(forBottleFolder: bottle.folderName)
        if !fm.fileExists(atPath: url.path) { fm.createFile(atPath: url.path, contents: nil) }
        let handle = try FileHandle(forWritingTo: url)
        try handle.seekToEnd()
        return handle
    }

    private func adjust(_ key: String, by delta: Int) {
        lock.withLock { counts[key, default: 0] += delta }
    }
}
```

```swift
import Foundation

/// Steam'in tarayıcı süreci (steamwebhelper) Wine'ın Mac sürücüsünde siyah pencere açıyor:
/// Chromium ayrı GPU sürecinden çiziyor, macdrv süreçler arası çizimi desteklemiyor.
/// Çözüm: gerçek dosyayı steamwebhelper_gercek.exe yapıp yerine onu --disable-gpu --single-process
/// ile açan aracıyı koymak. Steam güncellemesi aracıyı sildiğinde her açılışta yeniden yerleştirilir.
public enum SteamWebHelperProxy {
    public static let helperName = "steamwebhelper.exe"
    public static let realName = "steamwebhelper_gercek.exe"
    /// Aracı ~17 KB; gerçek dosya MB'larca. Bundan büyükse gerçek dosyadır.
    public static let realSizeThreshold = 500_000

    public static func cefDirectory(bottleURL: URL) -> URL {
        bottleURL.appendingPathComponent("drive_c/Program Files (x86)/Steam/bin/cef", isDirectory: true)
    }

    /// Bir şey değiştirdiyse true.
    @discardableResult
    public static func apply(bottleURL: URL, proxyExecutable: URL) throws -> Bool {
        let fm = FileManager.default
        let cef = cefDirectory(bottleURL: bottleURL)
        guard let children = try? fm.contentsOfDirectory(atPath: cef.path) else { return false }
        var changed = false
        for child in children.sorted() where child.hasPrefix("cef.win") {
            let dir = cef.appendingPathComponent(child, isDirectory: true)
            let helper = dir.appendingPathComponent(helperName)
            guard fm.fileExists(atPath: helper.path) else { continue }

            let size = (try fm.attributesOfItem(atPath: helper.path)[.size] as? NSNumber)?.intValue ?? 0
            if size > realSizeThreshold {
                let real = dir.appendingPathComponent(realName)
                if fm.fileExists(atPath: real.path) { try fm.removeItem(at: real) }
                try fm.moveItem(at: helper, to: real)
                changed = true
            }
            if !fm.contentsEqual(atPath: proxyExecutable.path, andPath: helper.path) {
                if fm.fileExists(atPath: helper.path) { try fm.removeItem(at: helper) }
                try fm.copyItem(at: proxyExecutable, to: helper)
                changed = true
            }
        }
        return changed
    }
}
```

```swift
import CryptoKit
import Foundation

public enum DownloadError: Error, Equatable, LocalizedError {
    case insecureNetwork
    case httpStatus(Int)
    case checksumMismatch(expected: String, actual: String)

    public var errorDescription: String? {
        switch self {
        case .insecureNetwork:
            "Ağ bağlantısı sunucuya güvenli şekilde ulaşamıyor. Kurum ağındaysan güvenlik duvarı engelliyor olabilir; başka bir ağda dene."
        case let .httpStatus(code):
            "Sunucu dosyayı vermedi (HTTP \(code))."
        case .checksumMismatch:
            "İndirilen dosya beklenen dosya değil (SHA-256 tutmadı); dosya silindi."
        }
    }
}

/// Kurulum dosyalarını indirir.
public struct Downloader: Sendable {
    public init() {}

    /// `source`'u `destination`'a indirir. `sha256` verilmişse doğrular ve hedefte doğru dosya
    /// zaten varsa tekrar indirmez.
    public func download(_ source: URL, to destination: URL, sha256: String?,
                         progress: @escaping @Sendable (Double) -> Void) async throws {
        let fm = FileManager.default
        let expected = sha256?.lowercased()
        if let expected, fm.fileExists(atPath: destination.path), (try? Self.sha256(of: destination)) == expected {
            progress(1)
            return
        }
        try fm.createDirectory(at: destination.deletingLastPathComponent(), withIntermediateDirectories: true)
        try await fetch(source, to: destination, progress: progress)
        if let expected {
            let actual = try Self.sha256(of: destination)
            guard actual == expected else {
                try? fm.removeItem(at: destination)
                throw DownloadError.checksumMismatch(expected: expected, actual: actual)
            }
        }
        progress(1)
    }

    private func fetch(_ source: URL, to destination: URL, progress: @escaping @Sendable (Double) -> Void) async throws {
        let box = ObservationBox()
        try await withCheckedThrowingContinuation { (continuation: CheckedContinuation<Void, any Error>) in
            let task = URLSession.shared.downloadTask(with: source) { temporary, response, error in
                box.invalidate()
                if let error {
                    continuation.resume(throwing: Self.map(error))
                    return
                }
                if let http = response as? HTTPURLResponse, !(200..<300).contains(http.statusCode) {
                    continuation.resume(throwing: DownloadError.httpStatus(http.statusCode))
                    return
                }
                guard let temporary else {
                    continuation.resume(throwing: URLError(.unknown))
                    return
                }
                do {
                    let fm = FileManager.default
                    if fm.fileExists(atPath: destination.path) { try fm.removeItem(at: destination) }
                    try fm.moveItem(at: temporary, to: destination)
                    continuation.resume()
                } catch {
                    continuation.resume(throwing: error)
                }
            }
            box.set(task.progress.observe(\.fractionCompleted) { value, _ in progress(value.fractionCompleted) })
            task.resume()
        }
    }

    /// Sertifika/TLS hataları (ör. kurum güvenlik duvarının TLS denetimi) → `insecureNetwork`.
    public static func map(_ error: any Error) -> any Error {
        guard let urlError = error as? URLError else { return error }
        switch urlError.code {
        case .serverCertificateUntrusted, .serverCertificateHasBadDate, .serverCertificateHasUnknownRoot,
             .serverCertificateNotYetValid, .secureConnectionFailed, .clientCertificateRejected, .clientCertificateRequired:
            return DownloadError.insecureNetwork
        default:
            return error
        }
    }

    public static func sha256(of file: URL) throws -> String {
        let handle = try FileHandle(forReadingFrom: file)
        defer { try? handle.close() }
        var hasher = SHA256()
        while let chunk = try handle.read(upToCount: 1 << 20), !chunk.isEmpty {
            hasher.update(data: chunk)
        }
        return hasher.finalize().map { String(format: "%02x", $0) }.joined()
    }
}

/// İlerleme gözlemcisini indirme bitene kadar canlı tutar.
private final class ObservationBox: @unchecked Sendable {
    private let lock = NSLock()
    private var observation: NSKeyValueObservation?

    func set(_ observation: NSKeyValueObservation) { lock.withLock { self.observation = observation } }

    func invalidate() {
        lock.withLock {
            observation?.invalidate()
            observation = nil
        }
    }
}
```

```swift
import Foundation

public enum InstallStep: String, CaseIterable, Hashable, Sendable {
    case preparing, downloading, installing, configuring

    public var title: String {
        switch self {
        case .preparing: "Bottle hazırlanıyor"
        case .downloading: "İndiriliyor"
        case .installing: "Kuruluyor"
        case .configuring: "Ayarlanıyor"
        }
    }
}

public enum InstallError: Error, Equatable, LocalizedError {
    case prefixInitFailed(exitCode: Int32)
    case installerFailed(exitCode: Int32)

    public var errorDescription: String? {
        switch self {
        case let .prefixInitFailed(code): "Windows ortamı hazırlanamadı (wineboot çıkış kodu \(code))."
        case let .installerFailed(code): "Kurulum programı hata ile bitti (çıkış kodu \(code))."
        }
    }
}

/// Kurulumun hangi adımda ve hangi bottle'da kaldığı.
public struct InstallFailure: Error, LocalizedError {
    public let step: InstallStep
    /// Kurulum sırasında oluşturulan (ya da kullanılan) bottle; ad hatasında nil.
    public let bottle: Bottle?
    public let underlying: any Error

    public init(step: InstallStep, bottle: Bottle?, underlying: any Error) {
        self.step = step
        self.bottle = bottle
        self.underlying = underlying
    }

    public var errorDescription: String? { underlying.localizedDescription }
}

/// Yeni bottle: klasör + kopru.json + wineboot.
public struct BottleFactory: Sendable {
    public let store: BottleStore
    public let runner: WineRunner

    public init(store: BottleStore, runner: WineRunner) {
        self.store = store
        self.runner = runner
    }

    public func create(name: String, settings: BottleSettings, recipeId: String?) async throws -> Bottle {
        let bottle = try store.makeBottle(name: name, settings: settings, recipeId: recipeId)
        // mscoree/mshtml kapalı: wineboot Wine Mono / Gecko indirme penceresi açmasın.
        let result = try await runner.run(.builtin("wineboot", arguments: ["--init"]), in: bottle,
                                          extraEnvironment: ["WINEDLLOVERRIDES": "mscoree=d;mshtml=d"])
        guard result.exitCode == 0 else {
            throw InstallFailure(step: .preparing, bottle: bottle, underlying: InstallError.prefixInitFailed(exitCode: result.exitCode))
        }
        return bottle
    }
}
```

`KopruCore/Sources/KopruCore/RecipeInstaller.swift`:

```swift
import Foundation

public enum InstallEvent: Equatable, Sendable {
    case started(InstallStep)
    /// Yalnızca indirme sırasında, 0...1
    case progress(Double)
    case finished(InstallStep)
    case bottleCreated(Bottle)
}

/// Tarifle kurulum: bottle → indir → kur → ayarla.
public struct RecipeInstaller: Sendable {
    public let factory: BottleFactory
    public let runner: WineRunner
    public let actions: RecipeActionPerformer
    public let downloader: Downloader
    public let downloadsDirectory: URL

    public init(factory: BottleFactory, runner: WineRunner, actions: RecipeActionPerformer,
                downloader: Downloader, downloadsDirectory: URL) {
        self.factory = factory
        self.runner = runner
        self.actions = actions
        self.downloader = downloader
        self.downloadsDirectory = downloadsDirectory
    }

    /// Hata her zaman `InstallFailure` olarak gelir.
    public func install(_ recipe: Recipe, bottleName: String,
                        onEvent: @escaping @Sendable (InstallEvent) -> Void) async throws -> Bottle {
        onEvent(.started(.preparing))
        let bottle: Bottle
        do {
            bottle = try await factory.create(name: bottleName, settings: recipe.bottle.settings, recipeId: recipe.id)
        } catch let failure as InstallFailure {
            throw failure
        } catch {
            throw InstallFailure(step: .preparing, bottle: nil, underlying: error)
        }
        onEvent(.bottleCreated(bottle))
        onEvent(.finished(.preparing))

        let installerFile = downloadsDirectory.appendingPathComponent(recipe.installer.fileName)
        try await step(.downloading, bottle, onEvent) {
            try await downloader.download(recipe.installer.url, to: installerFile, sha256: recipe.installer.sha256) {
                onEvent(.progress($0))
            }
        }
        try await step(.installing, bottle, onEvent) {
            let result = try await runner.run(.unixFile(installerFile, arguments: recipe.installer.arguments), in: bottle)
            guard result.exitCode == 0 else { throw InstallError.installerFailed(exitCode: result.exitCode) }
        }
        try await step(.configuring, bottle, onEvent) {
            try await actions.perform(recipe.afterInstall, recipe: recipe, bottle: bottle)
        }
        return bottle
    }

    private func step(_ step: InstallStep, _ bottle: Bottle, _ onEvent: @Sendable (InstallEvent) -> Void,
                      _ body: () async throws -> Void) async throws {
        onEvent(.started(step))
        do {
            try await body()
        } catch {
            throw InstallFailure(step: step, bottle: bottle, underlying: error)
        }
        onEvent(.finished(step))
    }
}
```

```swift
import AppKit
import KopruCore
import Observation

/// Ekranda gösterilecek hata ya da uyarı.
struct AppAlert: Identifiable {
    let id = UUID()
    let title: String
    let message: String
    var logURL: URL?
}

/// Açılışta Köprü'yü kullanılamaz yapan sorun.
enum StartupProblem: Equatable {
    case engineMissing(String)
    case rosettaMissing(String)
}

/// "Listede olmayan program"ın kurulacağı yer.
enum UnlistedTarget: Hashable {
    case newBottle(String)
    case existing(Bottle)
}

/// Kurulum ilerleme ekranının durumu.
struct InstallProgress: Identifiable {
    let id = UUID()
    let title: String
    let steps: [InstallStep]
    /// Yeni bottle açılıyorsa hata ekranı "yarım bottle'ı sil" sunar.
    let createsBottle: Bool
    var current: InstallStep?
    var done: Set<InstallStep> = []
    var fraction: Double?
    var failedStep: InstallStep?
    var errorMessage: String?
    var bottle: Bottle?
    var finished = false

    var isActive: Bool { !finished && failedStep == nil }
}

@MainActor
@Observable
final class AppModel {
    let paths: KopruPaths
    let store: BottleStore
    let catalog: RecipeCatalog
    private(set) var runner: WineRunner?
    private(set) var startupProblem: StartupProblem?

    private(set) var bottles: [Bottle] = []
    var selectedBottleID: Bottle.ID?
    private(set) var shortcuts: [ShortcutProgram] = []
    private(set) var runningCounts: [Bottle.ID: Int] = [:]
    var alert: AppAlert?
    var install: InstallProgress?

    init() {
        let paths = KopruPaths.standard
        self.paths = paths
        store = BottleStore(paths: paths)
        let resources = Bundle.main.resourceURL ?? Bundle.main.bundleURL
        let recipeFiles = resources.appendingPathComponent("RecipeFiles", isDirectory: true)
        catalog = (try? RecipeCatalog.load(recipesDirectory: resources.appendingPathComponent("Recipes", isDirectory: true),
                                           filesDirectory: recipeFiles))
            ?? RecipeCatalog(recipes: [], filesDirectory: recipeFiles)
        do {
            try Engine.checkRosetta()
            let engine = try Engine.locate(bundleResources: Bundle.main.resourceURL,
                                           environment: ProcessInfo.processInfo.environment,
                                           developmentFallback: Self.developmentFallback)
            runner = WineRunner(engine: engine, paths: paths)
        } catch EngineError.rosettaMissing {
            startupProblem = .rosettaMissing(EngineError.rosettaMissing.localizedDescription)
        } catch {
            startupProblem = .engineMissing(error.localizedDescription)
        }
        reload()
    }

    private static var developmentFallback: URL? {
        #if DEBUG
        return Engine.developmentEngine
        #else
        return nil
        #endif
    }

    // MARK: Durum

    var isReady: Bool { runner != nil }

    var selectedBottle: Bottle? { bottles.first { $0.id == selectedBottleID } }

    func recipe(for bottle: Bottle) -> Recipe? { catalog.recipe(id: bottle.info.recipeId) }

    var programs: [ProgramItem] {
        guard let bottle = selectedBottle else { return [] }
        return ProgramList.items(recipeLaunch: recipe(for: bottle)?.launch,
                                 pinned: bottle.info.pinnedPrograms,
                                 shortcuts: shortcuts)
    }

    func isRunning(_ bottle: Bottle) -> Bool { (runningCounts[bottle.id] ?? 0) > 0 }

    func isNameAvailable(_ name: String) -> Bool { (try? store.validateNewName(name)) != nil }

    func reload(select id: Bottle.ID? = nil) {
        do {
            bottles = try store.list()
        } catch {
            show("Bottle'lar okunamadı", error)
        }
        if let id { selectedBottleID = id }
        if selectedBottle == nil { selectedBottleID = bottles.first?.id }
        rescan()
    }

    func rescan() {
        guard let bottle = selectedBottle else {
            shortcuts = []
            return
        }
        let hidden = Set(bottle.info.hiddenShortcuts).union(recipe(for: bottle)?.hideShortcuts ?? [])
        shortcuts = ProgramScanner.scan(bottleURL: bottle.url, hidden: hidden)
    }

    // MARK: Programları açma

    func launch(_ item: ProgramItem) {
        guard let runner, let bottle = selectedBottle else { return }
        start(item.target, name: item.name, in: bottle, runner: runner)
    }

    func runExecutable(_ file: URL, pin: Bool) {
        guard let runner, var bottle = selectedBottle else { return }
        if pin {
            let program = PinnedProgram(name: file.deletingPathExtension().lastPathComponent,
                                        windowsPath: WindowsPath.fromUnix(file, bottleURL: bottle.url))
            if !bottle.info.pinnedPrograms.contains(program) {
                bottle.info.pinnedPrograms.append(program)
                save(bottle)
            }
        }
        start(.unixFile(file, arguments: []), name: file.lastPathComponent, in: bottle, runner: runner)
    }

    private func start(_ target: LaunchTarget, name: String, in bottle: Bottle, runner: WineRunner) {
        let recipe = recipe(for: bottle)
        let desktopSize = bottle.info.settings.virtualDesktop.size(visibleScreen: Self.visibleScreen())
        let actions = RecipeActionPerformer(runner: runner, catalog: catalog)
        runningCounts[bottle.id, default: 0] += 1
        Task {
            defer {
                runningCounts[bottle.id, default: 1] -= 1
                rescan()
            }
            do {
                if let recipe {
                    try await actions.perform(recipe.beforeLaunch, recipe: recipe, bottle: bottle)
                }
                let result = try await runner.run(target, in: bottle, desktopSize: desktopSize)
                if result.exitCode != 0 && result.duration < 5 {
                    alert = AppAlert(title: "\(name) hemen kapandı",
                                     message: "Program açıldıktan hemen sonra kapandı (çıkış kodu \(result.exitCode)). Ayrıntılar günlük dosyasında.",
                                     logURL: logURL(for: bottle))
                }
            } catch {
                show("\(name) açılamadı", error, log: bottle)
            }
        }
    }

    private static func visibleScreen() -> PixelSize {
        let frame = NSScreen.main?.visibleFrame ?? CGRect(x: 0, y: 0, width: 1640, height: 1040)
        return PixelSize(width: Int(frame.width), height: Int(frame.height))
    }

    func removeFromList(_ item: ProgramItem) {
        guard var bottle = selectedBottle else { return }
        switch item.source {
        case let .shortcut(shortcut):
            bottle.info.hiddenShortcuts.append(shortcut.relativePath)
        case let .pinned(program):
            bottle.info.pinnedPrograms.removeAll { $0 == program }
        case .recipe:
            return
        }
        save(bottle)
        rescan()
    }

    func reveal(_ item: ProgramItem) {
        guard let bottle = selectedBottle else { return }
        let url: URL? = switch item.source {
        case let .shortcut(shortcut): shortcut.url
        case let .pinned(program): WindowsPath.toUnix(program.windowsPath, bottleURL: bottle.url)
        case let .recipe(launch): WindowsPath.toUnix(launch.windowsPath, bottleURL: bottle.url)
        }
        if let url { NSWorkspace.shared.activateFileViewerSelecting([url]) }
    }

    private func shutdownTarget(for bottle: Bottle) -> LaunchTarget? {
        guard let launch = recipe(for: bottle)?.launch, let arguments = launch.shutdownArguments else { return nil }
        return .windowsPath(launch.windowsPath, arguments: arguments)
    }

    func stopSelected() {
        guard let runner, let bottle = selectedBottle else { return }
        let shutdown = shutdownTarget(for: bottle)
        Task { await runner.stop(bottle, shutdown: shutdown) }
    }

    // MARK: Kurulum

    func installRecipe(_ recipe: Recipe, bottleName: String) {
        guard let runner else { return }
        install = InstallProgress(title: "\(recipe.name) kuruluyor", steps: InstallStep.allCases, createsBottle: true)
        let installer = RecipeInstaller(factory: BottleFactory(store: store, runner: runner),
                                        runner: runner,
                                        actions: RecipeActionPerformer(runner: runner, catalog: catalog),
                                        downloader: Downloader(),
                                        downloadsDirectory: paths.downloads)
        Task {
            do {
                let bottle = try await installer.install(recipe, bottleName: bottleName) { event in
                    Task { @MainActor in self.apply(event) }
                }
                finishInstall(bottle)
            } catch {
                failInstall(error)
            }
        }
    }

    func installUnlisted(file: URL, target: UnlistedTarget) {
        guard let runner else { return }
        let createsBottle: Bool = if case .newBottle = target { true } else { false }
        install = InstallProgress(title: "\(file.lastPathComponent) kuruluyor",
                                  steps: createsBottle ? [.preparing, .installing] : [.installing],
                                  createsBottle: createsBottle)
        let factory = BottleFactory(store: store, runner: runner)
        Task {
            var bottle: Bottle?
            do {
                switch target {
                case let .existing(existing):
                    bottle = existing
                case let .newBottle(name):
                    apply(.started(.preparing))
                    let created = try await factory.create(name: name, settings: .default, recipeId: nil)
                    bottle = created
                    apply(.bottleCreated(created))
                    apply(.finished(.preparing))
                }
                guard let bottle else { return }
                apply(.started(.installing))
                let result = try await runner.run(.unixFile(file, arguments: []), in: bottle)
                guard result.exitCode == 0 else {
                    throw InstallFailure(step: .installing, bottle: bottle,
                                         underlying: InstallError.installerFailed(exitCode: result.exitCode))
                }
                finishInstall(bottle)
            } catch let failure as InstallFailure {
                failInstall(failure)
            } catch {
                failInstall(InstallFailure(step: install?.current ?? .preparing, bottle: bottle, underlying: error))
            }
        }
    }

    func createEmptyBottle(name: String) {
        guard let runner else { return }
        install = InstallProgress(title: "\(name) oluşturuluyor", steps: [.preparing], createsBottle: true)
        Task {
            do {
                apply(.started(.preparing))
                let bottle = try await BottleFactory(store: store, runner: runner).create(name: name, settings: .default, recipeId: nil)
                finishInstall(bottle)
            } catch {
                failInstall(error)
            }
        }
    }

    private func apply(_ event: InstallEvent) {
        guard var progress = install, progress.isActive else { return }
        switch event {
        case let .started(step):
            progress.current = step
            progress.fraction = nil
        case let .progress(fraction):
            progress.fraction = fraction
        case let .finished(step):
            progress.done.insert(step)
            if progress.current == step { progress.current = nil }
        case let .bottleCreated(bottle):
            progress.bottle = bottle
        }
        install = progress
        if case let .bottleCreated(bottle) = event { reload(select: bottle.id) }
    }

    private func finishInstall(_ bottle: Bottle) {
        if var progress = install {
            progress.done = Set(progress.steps)
            progress.current = nil
            progress.finished = true
            progress.bottle = bottle
            install = progress
        }
        reload(select: bottle.id)
    }

    private func failInstall(_ error: any Error) {
        guard var progress = install else { return }
        let failure = error as? InstallFailure
        progress.failedStep = failure?.step ?? progress.current ?? progress.steps.first
        progress.errorMessage = error.localizedDescription
        if progress.bottle == nil, progress.createsBottle { progress.bottle = failure?.bottle }
        progress.current = nil
        install = progress
        reload()
    }

    func deleteFailedInstallBottle() {
        if let progress = install, progress.createsBottle, let bottle = progress.bottle {
            deleteBottle(bottle)
        }
        install = nil
    }

    // MARK: Bottle yönetimi

    func importBottle(from folder: URL, name: String) {
        guard let runner else { return }
        Task {
            // Taşımadan önce o bottle'da çalışan Windows süreçleri kapatılır.
            await runner.killWineserver(prefix: folder, settings: .default)
            do {
                let bottle = try store.importBottle(from: folder, name: name, catalog: catalog)
                reload(select: bottle.id)
            } catch {
                show("Bottle içe aktarılamadı", error)
            }
        }
    }

    func update(_ bottle: Bottle, name: String, settings: BottleSettings) {
        guard let runner else { return }
        let shutdown = shutdownTarget(for: bottle)
        Task {
            do {
                var updated = bottle
                if updated.info.name != name.trimmingCharacters(in: .whitespacesAndNewlines) {
                    updated = try store.rename(updated, to: name)
                }
                // Bir bottle'daki bütün süreçler aynı MSync ayarıyla başlamalı; değişince bottle durdurulur.
                if updated.info.settings.msync != settings.msync {
                    await runner.stop(updated, shutdown: shutdown)
                }
                updated.info.settings = settings
                try store.save(updated)
                reload(select: updated.id)
            } catch {
                show("Ayarlar kaydedilemedi", error)
            }
        }
    }

    func deleteBottle(_ bottle: Bottle) {
        let shutdown = shutdownTarget(for: bottle)
        Task {
            if let runner { await runner.stop(bottle, shutdown: shutdown) }
            do {
                try store.delete(bottle)
                if selectedBottleID == bottle.id { selectedBottleID = nil }
                reload()
            } catch {
                show("Bottle silinemedi", error)
            }
        }
    }

    private func save(_ bottle: Bottle) {
        do {
            try store.save(bottle)
            if let index = bottles.firstIndex(where: { $0.id == bottle.id }) { bottles[index] = bottle }
        } catch {
            show("Bottle kaydedilemedi", error)
        }
    }

    // MARK: Yardımcılar

    func openDriveC() {
        if let bottle = selectedBottle { NSWorkspace.shared.open(bottle.driveC) }
    }

    func openWineSettings(_ bottle: Bottle) {
        guard let runner else { return }
        start(.builtin("winecfg", arguments: []), name: "Windows Ayarları", in: bottle, runner: runner)
    }

    func logURL(for bottle: Bottle) -> URL { paths.logFile(forBottleFolder: bottle.folderName) }

    func openLog(_ url: URL) { NSWorkspace.shared.open(url) }

    private func show(_ title: String, _ error: any Error, log bottle: Bottle? = nil) {
        alert = AppAlert(title: title, message: error.localizedDescription, logURL: bottle.map(logURL(for:)))
    }
}
```

```swift
                                .monospacedDigit()

                }

                Text(message).foregroundStyle(.red).textSelection(.enabled)

            HStack {
                if progress.failedStep != nil {
                    if let bottle = progress.bottle {
                        Button("Günlüğü Göster") { model.openLog(model.logURL(for: bottle)) }

                        .keyboardShortcut(.defaultAction)
                } else if progress.finished {
                    Spacer()
                    Button("Bitti") { model.install = nil }
                        .keyboardShortcut(.defaultAction)
                } else {
                    Spacer()
                    Text("Kurulum sürüyor…").foregroundStyle(.secondary)
                }
            }
        }
    }

    private func icon(for step: InstallStep, in progress: InstallProgress) -> some View {

            Image(systemName: "checkmark.circle.fill").foregroundStyle(.green)

        } else {
            Image(systemName: "circle").foregroundStyle(.tertiary)
        }
    }
```

```swift
import KopruCore
import SwiftUI
import UniformTypeIdentifiers

struct ContentView: View {
    @Environment(AppModel.self) private var model
    @State private var installStart: InstallSheet.Start?
    @State private var showSettings = false
    @State private var pickExecutable = false
    @State private var executableToRun: URL?

    var body: some View {
        @Bindable var model = model
        NavigationSplitView {
            BottleSidebar()
                .navigationSplitViewColumnWidth(min: 180, ideal: 210)
        } detail: {
            detail
        }
        .toolbar { toolbar }
        .onChange(of: model.selectedBottleID) { model.rescan() }
        .dropDestination(for: URL.self) { urls, _ in
            guard model.isReady, let file = urls.first,
                  ["exe", "msi"].contains(file.pathExtension.lowercased()) else { return false }
            installStart = .unlisted(file)
            return true
        }
        .sheet(item: $installStart) { start in
            InstallSheet(start: start)
        }
        .sheet(item: progressBinding) { _ in
            InstallProgressView()
        }
        .sheet(isPresented: $showSettings) {
            if let bottle = model.selectedBottle {
                BottleSettingsSheet(bottle: bottle)
            }
        }
        .fileImporter(isPresented: $pickExecutable, allowedContentTypes: UTType.windowsInstallers) { result in
            if case let .success(url) = result { executableToRun = url }
        }
        .fileDialogDefaultDirectory(model.selectedBottle?.driveC)
        .confirmationDialog("Program çalıştırılsın mı?",
                            isPresented: Binding(get: { executableToRun != nil }, set: { if !$0 { executableToRun = nil } }),
                            presenting: executableToRun) { url in
            Button("Çalıştır") { model.runExecutable(url, pin: false) }
            Button("Çalıştır ve listeye ekle") { model.runExecutable(url, pin: true) }
            Button("Vazgeç", role: .cancel) {}
        } message: { url in
            Text(url.lastPathComponent)
        }
        .alert(model.alert?.title ?? "",
               isPresented: Binding(get: { model.alert != nil }, set: { if !$0 { model.alert = nil } }),
               presenting: model.alert) { alert in
            if let log = alert.logURL {
                Button("Günlüğü Göster") { model.openLog(log) }
            }
            Button("Tamam", role: .cancel) {}
        } message: { alert in
            Text(alert.message)
        }
    }

    /// İlerleme penceresi yalnızca Kur penceresi kapalıyken gösterilir (Kur'a basınca o kapanır, bu açılır).
    private var progressBinding: Binding<InstallProgress?> {
        Binding(get: { installStart == nil ? model.install : nil },
                set: { if $0 == nil { model.install = nil } })
    }

    @ViewBuilder
    private var detail: some View {
        if let problem = model.startupProblem {
            StartupProblemView(problem: problem)
        } else if let bottle = model.selectedBottle {
            ProgramGrid(bottle: bottle)
        } else {
            ContentUnavailableView {
                Label("Henüz bottle yok", systemImage: "shippingbox")
            } description: {
                Text("Kur'a bas ya da bir kurulum dosyasını (.exe, .msi) bu pencereye sürükle.")
            } actions: {
                Button("Program Kur") { installStart = .catalog }
                    .disabled(!model.isReady)
            }
        }
    }

    @ToolbarContentBuilder
    private var toolbar: some ToolbarContent {
        ToolbarItemGroup(placement: .primaryAction) {
            Button { installStart = .catalog } label: { Label("Kur", systemImage: "plus.circle") }
                .disabled(!model.isReady)
            Button { model.openDriveC() } label: { Label("C: Sürücüsü", systemImage: "internaldrive") }
                .disabled(model.selectedBottle == nil)
            Button { pickExecutable = true } label: { Label("Program Çalıştır…", systemImage: "play") }
                .disabled(model.selectedBottle == nil || !model.isReady)
            Button { model.stopSelected() } label: { Label("Durdur", systemImage: "stop.circle") }
                .disabled(model.selectedBottle == nil || !model.isReady)
            Button { showSettings = true } label: { Label("Ayarlar", systemImage: "gearshape") }
                .disabled(model.selectedBottle == nil)
        }
    }
}
```
